\begin{frame}{2. Thách thức khoa học: Hiện tượng Lan truyền sai số}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{\textbf{Cơ chế phát sinh \& khuếch đại sai số}}
        \footnotesize
        \begin{itemize}
          \item \textbf{Sai số nhận dạng thị giác:} Do che khuất, số đếm từ camera $f_{\theta_1}$ luôn chứa sai số $\boldsymbol{\eta}(t)$:
            \begin{equation*}
              \mathbf{x}_i(t) = c_i(t) + \boldsymbol{\eta}_i(t)
            \end{equation*}
          \item \textbf{Lan truyền theo không gian:} Phép tích chập đồ thị ChebNet khuếch tán sai số $\boldsymbol{\eta}$ sang các nút lân cận $K$-hop.
          \item \textbf{Tích lũy theo thời gian:} Khi dự báo đa bước $H=6$ (30 phút), sai số ngắn hạn làm méo xu hướng dài hạn.
        \end{itemize}
      \end{alertblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Sơ đồ chuỗi lan truyền sai số nối tầng}}

\begin{frame}{7. Giai đoạn 2: Kiến trúc mô hình đề xuất TA-STGCN}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{ptitblue}{Cấu trúc 3 khối xử lý không -- thời gian:}}
      \begin{enumerate}
        \footnotesize
        \item \textbf{Tích chập không gian ChebNet ($K=3$):}
          \begin{equation*}
            g_\theta \star_{\mathcal{G}} \mathbf{x} = \sum_{k=0}^{K-1} \theta_k T_k(\tilde{\mathbf{L}}) \mathbf{x}
          \end{equation*}
          Lan truyền thông tin lân cận $K$-hop, độ phức tạp $\mathcal{O}(K|\mathcal{E}|)$.
        \item \textbf{Tích chập thời gian có cổng (1D GLU):}
          Huấn luyện song song cửa sổ $T_{\text{in}}=24$, khắc phục nghẽn tuần tự của RNN/GRU.
        \item \textbf{Cơ chế chú ý thời gian đa đầu (MHTA):}
          Xen kẽ giữa 2 khối ST-Conv để tái cân bằng trọng số giữa các mốc thời gian lịch sử.
      \end{enumerate}
      \vskip 2pt
      \badge{ptitblue}{Số tham số: \textbf{453K}} \quad
      \badge{ptitgreen}{VRAM: \textbf{1.6 GB}}
    \end{column}

% SLIDE 9: CƠ CHẾ KHỬ NHIỄU CỦA TEMPORAL ATTENTION

\begin{frame}{8. Cơ chế khử nhiễu của Temporal Self-Attention}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{alertblock}{\textbf{Nguyên lý toán học: Bộ lọc thông thấp thích ứng}}
        \footnotesize
        Khi đầu vào chứa sai số nhận dạng $\boldsymbol{\eta} \sim (\mu, \sigma^2)$, tầng Temporal Attention tính toán tổ hợp lồi:
        \begin{equation*}
          \mathbf{h}_t = \sum_{\tau=1}^{T} \alpha_{t,\tau} \mathbf{v}_\tau \implies \operatorname{Var}(\mathbf{h}_t) \approx \frac{\sigma_\eta^2}{T_{\text{eff}}}
        \end{equation*}
        $\to$ \textbf{Phương sai sai số giảm tỷ lệ nghịch với cửa sổ chú ý hiệu dụng $T_{\text{eff}}$}!
      \end{alertblock}

      \begin{block}{\textbf{Vai trò thực nghiệm}}
        \scriptsize
        \begin{itemize}
          \item Tự động nhận diện các mốc thời gian có tính chu kỳ cao.
          \item Triệt tiêu các đột biến do lỗi nhận dạng camera cục bộ gây ra.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=4.4cm,keepaspectratio]{paper/fig/attention_heatmap_morning_peak.png}
      \vskip 2pt
      {\tiny \textbf{Hình: Bản đồ nhiệt phân bổ trọng số chú ý trong giờ cao điểm sáng}}
      \vskip 3pt
      \begin{beamercolorbox}[rounded=true,wd=\linewidth,sep=3pt]{block body example}
        \scriptsize
        \textbf{Nhận xét:} Trọng số tập trung mạnh vào các bước quan sát phản ánh đúng nhịp tăng lưu lượng trong giờ cao điểm.
      \end{beamercolorbox}
    \end{column}
  \end{columns}
\end{frame}
