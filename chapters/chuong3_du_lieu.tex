\chapter{Thu thập dữ liệu và xây dựng đồ thị giao thông (IC4SD-Traffic-HCM)}
\label{chap:du_lieu}

Để phục vụ bài toán dự báo dòng giao thông hỗn hợp tại các đô thị Đông Nam Á và đánh giá một cách có hệ thống hiện tượng lan truyền sai số nhận dạng, nghiên cứu đã tiến hành thu thập, chuẩn hóa và xây dựng bộ dữ liệu quy mô đô thị mang tên \textbf{IC4SD-Traffic-HCM}. Bộ dữ liệu tích hợp hai thành phần: (1) Bộ dữ liệu hình ảnh gán nhãn chi tiết phục vụ huấn luyện và kiểm định mô hình đếm xe ở Giai đoạn 1; (2) Bộ dữ liệu chuỗi thời gian liên tục kết hợp đồ thị đường bộ thực tế phục vụ bài toán dự báo mạng lưới ở Giai đoạn 2.

\section{Hạ tầng camera giám sát và hệ thống thu thập tự động}
\label{sec:ha_tang_camera}

\subsection{Hạ tầng camera giám sát TP. Hồ Chí Minh}
Thành phố Hồ Chí Minh là trung tâm kinh tế lớn nhất cả nước với diện tích hơn 2.061 $\text{km}^2$ và dân số hơn 10 triệu người. Mạng lưới đường bộ thành phố chịu tải lượng lớn từ hơn 8.5 triệu xe máy cá nhân và gần 1 triệu xe ô tô. Để phục vụ công tác điều hành giao thông, Sở Giao thông Vận tải TP.HCM đã triển khai hệ thống camera giám sát giao thông (CCTV) bao phủ các trục đường chính, cầu vượt, hầm ngầm và các nút giao thông trọng điểm.

Dữ liệu hình ảnh từ các camera này được trích xuất từ Cổng thông tin giao thông thành phố Hồ Chí Minh dưới dạng ảnh tĩnh định kỳ, phục vụ giám sát và nghiên cứu khoa học.

\subsection{Hệ thống thu thập dữ liệu đa luồng tự động}
Nhóm nghiên cứu đã phát triển hệ thống thu thập dữ liệu tự động vận hành liên tục trên máy chủ chuyên dụng, thực hiện thu thập dữ liệu ảnh đa luồng bất đồng bộ và sàng lọc các điểm trạm camera giám sát ổn định theo Thuật toán \ref{algo:data_collection}.
\begin{algorithm_box}{Thu thập dữ liệu ảnh đa luồng bất đồng bộ và sàng lọc trạm camera}{
\textbf{Đầu vào:} Danh mục ban đầu $M=657$ trạm camera $\mathcal{C}_{\text{init}} = \{c_1, c_2, \dots, c_M\}$, chu kỳ lấy mẫu $\tau = 60$ giây, thời gian thử nghiệm $T_{\text{test}} = 14$ ngày, ngưỡng lỗi tối đa $\theta_{\text{err}} = 5\%$. \\
\textbf{Đầu ra:} Tập $N=608$ trạm camera ổn định $\mathcal{V}$ và kho lưu trữ hình ảnh theo chu kỳ $\mathcal{D}_{\text{raw}}$. \\
\textbf{Bước 1: Khởi tạo kết nối đa luồng bất đồng bộ:}
\begin{enumerate}
    \item Khởi tạo nhóm phiên làm việc HTTP bất đồng bộ (Session Pool) với cơ chế giữ kết nối (Keep-Alive) nhằm triệt tiêu độ trễ bắt tay TCP/TLS.
    \item Thiết lập biến đếm số lần yêu cầu $S_k \gets 0$ và số lần ghi nhận lỗi $E_k \gets 0$ cho từng trạm $c_k \in \mathcal{C}_{\text{init}}$.
\end{enumerate}
\textbf{Bước 2: Vòng lặp thu thập đồng bộ theo chu kỳ $\tau$:}
\begin{enumerate}
    \item \textbf{Định kỳ} mỗi $\tau = 60$ giây:
    \begin{itemize}
        \item Kích hoạt đồng thời $M$ tác vụ yêu cầu bất đồng bộ (Async HTTP GET) tới các endpoint của $M$ trạm camera.
        \item Thu nhận phản hồi trong cửa sổ $2 \div 4$ giây:
        \item \textbf{Với mỗi} phản hồi từ trạm $c_k$:
        \begin{itemize}
            \item Cập nhật $S_k \gets S_k + 1$.
            \item Kiểm tra tính toàn vẹn: nếu mã phản hồi là HTTP 200, kích thước tệp hợp lệ và không phải khung hình đen hoàn toàn thì ghi tệp ảnh vào $\mathcal{D}_{\text{raw}}$ kèm nhãn thời gian $t$.
            \item Ngược lại (mã lỗi 404, 503, timeout hoặc ảnh rỗng), cập nhật $E_k \gets E_k + 1$.
        \end{itemize}
    \end{itemize}
    \item Lặp lại quy trình cho đến khi đủ thời gian quan sát $T_{\text{test}}$.
\end{enumerate}
\textbf{Bước 3: Sàng lọc trạm ổn định:}
\begin{enumerate}
    \item Với mỗi trạm $c_k$, tính tỷ lệ mất tín hiệu tích lũy: $r_k = E_k / S_k$.
    \item Lọc danh sách trạm chính thức: $\mathcal{V} = \{c_k \in \mathcal{C}_{\text{init}} \mid r_k \le \theta_{\text{err}}\}$.
    \item Thu được đúng $N = |\mathcal{V}| = 608$ trạm camera hoạt động liên tục và ổn định.
\end{enumerate}
}
\label{algo:data_collection}
\end{algorithm_box}

Hình \ref{fig:anh_samples_camera_chapter3} minh họa một số mẫu hình ảnh thu thập thực tế từ hệ thống, cho thấy sự đa dạng về góc chụp (từ trên cao nhìn xuống, góc chéo ngang mép đường), điều kiện ánh sáng (ban ngày, chạng vạng hoàng hôn, ban đêm) và mật độ tập trung cao của luồng xe máy.

\section{Bộ dữ liệu hình ảnh IC4SD-Traffic-HCM-Vision}
\label{sec:du_lieu_vision}

Từ kho dữ liệu thu thập được, nhóm nghiên cứu đã trích xuất một tập đại diện gồm \textbf{5.012 ảnh chụp độ phân giải cao} ($1920 \times 1080$ hoặc $1280 \times 720$) để xây dựng bộ dữ liệu thị giác IC4SD-Traffic-HCM-Vision.

\subsection{Giao thức gán nhãn độc lập ba thẩm định viên}
Để giảm thiểu sai lệch chủ quan của con người khi đếm các cảnh xe máy đông đúc, quy trình gán nhãn được tổ chức theo quy chuẩn thống nhất:
\begin{itemize}
    \item Ba thẩm định viên nghiên cứu độc lập (được tập huấn thống nhất về định nghĩa phương tiện và quy tắc nhận diện xe bị che khuất) tiến hành gán nhãn toàn bộ 5.012 ảnh trên phần mềm chuyên dụng. Mỗi ảnh được gán nhãn độc lập bởi cả ba người.
    \item Định nghĩa hai lớp phương tiện chính:
    \begin{enumerate}
        \item \textbf{Xe ô tô con:} Xe ô tô dưới 9 chỗ, xe taxi, xe SUV và xe bán tải đang lưu thông hoặc dừng chờ đèn tín hiệu trên mặt đường quan sát được.
        \item \textbf{Xe máy:} Tất cả các loại xe mô tô hai bánh, xe tay ga, xe số và xe máy điện -- phương tiện chủ đạo trong dòng giao thông đô thị Việt Nam.
    \end{enumerate}
    \item Các ảnh có độ lệch số đếm giữa ba người vượt quá $5\%$ đối với ô tô hoặc vượt quá $10\%$ đối với xe máy được đưa vào phiên thẩm tra chung. Tại đây, cả ba chuyên viên cùng rà soát hình ảnh, đối chiếu từng cụm xe nghi vấn và thống nhất nhãn thực tế cuối cùng.
\end{itemize}

\subsection{Đánh giá độ tin cậy gán nhãn bằng hệ số tương quan nội lớp (ICC)}
Để định lượng độ tin cậy và sự thống nhất của dữ liệu nhãn trước khi thẩm tra chung, nhóm nghiên cứu tính toán hệ số tương quan nội lớp (ICC) trên một tập mẫu ngẫu nhiên gồm 500 ảnh (xấp xỉ $10\%$ dữ liệu). 

Mô hình được áp dụng là $\text{ICC}(3, k)$ -- mô hình hiệu ứng hỗn hợp hai chiều đánh giá mức độ đồng thuận tuyệt đối dựa trên giá trị trung bình của $k$ người đánh giá ($k=3$):
\begin{equation}
    \text{ICC}(3, k) = \frac{\text{MS}_R - \text{MS}_E}{\text{MS}_R}
\end{equation}
trong đó $\text{MS}_R$ là phương sai trung bình giữa các ảnh và $\text{MS}_E$ là phương sai trung bình của sai số dư.

\textbf{Kết quả đo đạc thống kê:}
\begin{itemize}
    \item Đối với lớp ô tô con: $\text{ICC} = 0.984$ (khoảng tin cậy 95\%: $0.981 - 0.987, p < 0.001$).
    \item Đối với lớp xe máy: $\text{ICC} = 0.962$ (khoảng tin cậy 95\%: $0.956 - 0.968, p < 0.001$).
    \item Tổng thể kết hợp cả hai lớp: $\text{ICC} = 0.973$ ($p < 0.001$).
\end{itemize}
Theo các chuẩn mực thống kê, chỉ số ICC trên $0.90$ phản ánh độ tin cậy rất cao. Phân tích theo các điều kiện lưu thông cho thấy ở điều kiện đường vắng vào ban đêm, chỉ số ICC đạt tới $0.995$. Trong điều kiện ùn ứ vào giờ cao điểm, chỉ số ICC của xe máy giảm nhẹ xuống $0.941$, phản ánh sự che khuất khách quan khi các xe máy đi sát nhau chỉ để lộ một phần thân xe hoặc mũ bảo hiểm.

\subsection{Đặc tính thống kê và phân bố theo khung giờ trong ngày}
Dữ liệu nhãn thể hiện tính đa dạng:
\begin{itemize}
    \item Trung bình số lượng phương tiện trên một ảnh là $30.28$ xe ($\text{Std} = 25.07$), giá trị trung vị là 24 xe, biên độ phân bố từ 0 xe (đường vắng lúc rạng sáng) lên tới 194 xe (ngã tư giờ cao điểm chiều).
    \item Lớp xe máy chiếm ưu thế với trung bình $23.02$ xe/ảnh (cao nhất 178 xe máy trong một khung hình), trong khi ô tô con đạt trung bình $7.26$ xe/ảnh (cao nhất 50 xe).
    \item Dữ liệu có phân bố lệch phải, phản ánh đúng quy luật biến động của giao thông đô thị thực tế.
\end{itemize}

\subsection{Kỹ thuật tăng cường dữ liệu bảo toàn nhãn}
Vì mục tiêu là hồi quy chính xác số lượng phương tiện có trong ảnh, các phương pháp cắt ngẫu nhiên (Random Crop) truyền thống sẽ làm mất một phần phương tiện ở rìa ảnh khiến nhãn bị sai lệch. Nghiên cứu thiết kế quy trình tăng cường dữ liệu bảo toàn nhãn:
\begin{enumerate}
    \item \textbf{Lật ngang ngẫu nhiên (Random Horizontal Flip):} Áp dụng với xác suất $p = 0.5$. Hướng di chuyển của xe đảo chiều nhưng tổng số lượng ô tô và xe máy được bảo toàn nguyên vẹn.
    \item \textbf{Biến đổi màu sắc trắc quang (Photometric Color Jittering):} Điều chỉnh ngẫu nhiên 4 thông số: độ sáng ($\pm 15\%$), độ tương phản ($\pm 15\%$), độ bão hòa màu ($\pm 15\%$) và sắc độ ($\pm 15\%$). Phép biến đổi này mô phỏng sự biến thiên của điều kiện thời tiết (nắng gắt, trời râm mát, bóng râm tòa nhà) mà không làm biến dạng hình học của phương tiện.
    \item \textbf{Xóa ngẫu nhiên diện tích nhỏ (Random Erasing):} Áp dụng với xác suất $p = 0.2$, tỷ lệ diện tích bị xóa từ $2\%$ đến $20\%$ bề mặt ảnh. Kỹ thuật này giúp mô hình không phụ thuộc vào một chi tiết đơn lẻ, tăng khả năng nhận diện xe khi bị cành cây ven đường hoặc biển báo giao thông che khuất một phần.
\end{enumerate}

\section{Bộ dữ liệu chuỗi thời gian IC4SD-Traffic-HCM-Graph}
\label{sec:du_lieu_graph}

Để phục vụ bài toán dự báo đồ thị ở Giai đoạn 2, mô hình thị giác tối ưu ở Giai đoạn 1 (EfficientNet-B5 \cite{efficientnet}) được triển khai để suy luận trên toàn bộ ảnh của 608 trạm camera trong suốt \textbf{86 ngày liên tục} (từ ngày 23/06/2025 đến ngày 16/09/2025). 

\subsection{Quy trình lọc nhân quả kết hợp trượt kép}
Dữ liệu suy luận ở tần suất 1 phút phản ánh lưu lượng tức thời tại thời điểm chụp ảnh. Do góc chụp camera và chu kỳ đèn tín hiệu, chuỗi 1 phút xuất hiện các dao động tần số cao. Nghiên cứu áp dụng quy trình lọc nhân quả hai bước:
\begin{enumerate}
    \item \textbf{Lọc trung bình trượt hồi quy (Backward Rolling Mean):} Sử dụng cửa sổ trượt 3 phút nhìn ngược về quá khứ ($window_1 = 3$, $center = \text{False}$):
    \begin{equation}
        s_{1}(t) = \frac{1}{3} \sum_{k=0}^{2} y(t - k)
    \end{equation}
    \item \textbf{Làm mượt hàm mũ xuôi chiều (Forward-in-time EMA):} Áp dụng làm mượt hàm mũ với tham số $span = 5$ (tương đương hệ số làm mượt $\alpha = 2/(5+1) \approx 0.333$):
    \begin{equation}
        s_{2}(t) = \alpha s_{1}(t) + (1 - \alpha) s_{2}(t - 1)
    \end{equation}
\end{enumerate}
Chuỗi sau khi làm mượt được lấy mẫu lại định kỳ đúng 5 phút một lần ($\Delta t = 5$ phút). Quy trình này đảm bảo tính nhân quả: giá trị làm mượt tại thời điểm $t$ không phụ thuộc vào bất kỳ điểm dữ liệu nào trong tương lai $t+1, t+2, \dots$

Kết quả thu được là tensor dữ liệu chuỗi thời gian hoàn chỉnh có kích thước:
\begin{equation}
    \mathbf{Y}_{\text{total}} \in \mathbb{R}^{24768 \times 608 \times 2}
\end{equation}
bao gồm đúng 24.768 bước thời gian liên tục, phủ kín 608 nút giao thông với 2 kênh lưu lượng phương tiện độc lập.

\section{Xây dựng đồ thị giao thông và ma trận kề RBF từ OpenStreetMap}
\label{sec:xay_dung_do_thi_osm}

\subsection{Trích xuất cạnh đồ thị theo tuyến đường thực tế}
Để đồ thị phản ánh đúng các ràng buộc vật lý của mạng lưới đường bộ thành phố (như các tuyến đường một chiều, dải phân cách cứng, cầu vượt hoặc sông ngòi kênh rạch), nhóm nghiên cứu trích xuất dữ liệu bản đồ OpenStreetMap (OSM) của TP. Hồ Chí Minh.

Một cạnh có hướng $(i, j) \in \mathcal{E}$ chỉ được thiết lập nếu tồn tại hành lang giao thông đường bộ hợp lệ cho phép phương tiện di chuyển trực tiếp từ vị trí camera $i$ đến vị trí camera $j$ theo đúng chiều quy định của luật giao thông mà không đi qua camera quan sát trung gian nào khác. Tổng cộng có \textbf{2.452 cạnh có hướng hợp lệ} được thiết lập trên 608 đỉnh.

\subsection{Xác định trọng số ma trận kề RBF}
Với mỗi cặp nút có kết nối hợp lệ $(i, j) \in \mathcal{E}$, khoảng cách định tuyến đường bộ ngắn nhất $d_{ij}$ (tính bằng kilomet) được tính toán qua thuật toán Dijkstra trên đồ thị đường phố OSM. Trọng số liên kết $W_{ij} \in [0, 1]$ được tính toán bằng hàm cơ sở xuyên tâm (RBF):
\begin{equation}
W_{ij} =
\begin{cases}
\exp\left(-\frac{d_{ij}^2}{\sigma_{\mathrm{RBF}}^2}\right), & \text{nếu } (i,j) \in \mathcal{E} \text{ và } d_{ij} \le \tau, \\
0, & \text{ngược lại}
\end{cases}
\end{equation}
trong đó:
\begin{itemize}
    \item $\sigma_{\mathrm{RBF}} = 1.0887\text{ km}$ là độ lệch chuẩn thực nghiệm của khoảng cách các cạnh kết nối hợp lệ trong mạng lưới đường phố thành phố.
    \item $\tau = 3.3945\text{ km}$ là ngưỡng bán kính cắt lọc không gian, đảm bảo tính thưa của ma trận kề và loại bỏ các liên kết quá xa vốn có tương quan mờ nhạt.
\end{itemize}

\subsection{Chuẩn hóa toán tử Laplacian Chebyshev có hướng}
Ma trận trọng số $\mathbf{W}$ được chuẩn hóa bậc hai phía nhằm bảo toàn cấu trúc luồng xe một chiều:
\begin{equation}
    \mathbf{L} = \mathbf{I}_N - \mathbf{D}^{-1/2} \mathbf{W} \mathbf{D}^{-1/2}, \quad \tilde{\mathbf{L}} = \frac{2}{\lambda_{\max}} \mathbf{L} - \mathbf{I}_N
\end{equation}
với $D_{ii} = \sum_j W_{ij}$ là ma trận bậc ra và $\lambda_{\max} \approx 2.0$ là giá trị riêng cực đại dùng để co giãn toán tử phục vụ khai triển trực giao Chebyshev \cite{chebnet}.

\section{Chiến lược phân chia dữ liệu và kiểm soát rò rỉ thông tin}

Điều này đảm bảo đánh giá chính xác khả năng tổng quát hóa của backbone trước các góc nhìn mới.
    \item \textbf{Phân lập thời gian cho Giai đoạn 2:} Chuỗi 24.768 bước thời gian được chia tuần tự theo dòng thời gian thực tế: $80\%$ đầu tiên làm tập Train (19.814 bước), $10\%$ tiếp theo làm tập Validation (2.477 bước) và $10\%$ cuối cùng làm tập Test (2.477 bước).

Các thông số $\mu_i$ và $\sigma_i$ của từng nút $i$ được tính toán duy nhất trên tập Train (80\% đầu) và đóng băng hoàn toàn khi áp dụng để chuẩn hóa tập Validation và Test.
\end{enumerate}

\section{Đặc tính chuỗi thời gian và biến động ngoại sinh trong 86 ngày}
\label{sec:dac_tinh_chuoi_thoi_gian}

Chuỗi thời gian liên tục 86 ngày ghi nhận đầy đủ các quy luật vận động của giao thông TP. Hồ Chí Minh:
\begin{itemize}
    \item \textbf{Chu kỳ ngày đêm:} Hình thái lưu lượng thể hiện cấu trúc hai đỉnh rõ nét:
    \begin{itemize}
        \item \textit{Đỉnh cao điểm sáng (07:00 -- 08:30):} Lưu lượng xe máy tăng cao từ $300\%$ đến $500\%$ so với mức trung bình ban ngày, tập trung từ các khu dân cư và đô thị vệ tinh hướng về trung tâm Quận 1 và Quận 3.
        \item \textit{Đỉnh cao điểm chiều (17:00 -- 18:30):} Luồng phương tiện đảo chiều di chuyển ra các cửa ngõ thành phố với thời gian giải tỏa kéo dài hơn do người dân kết hợp sinh hoạt cá nhân.
        \item \textit{Đáy ban đêm (00:00 -- 05:00):} Lưu lượng xe máy giảm xuống dưới $5\%$ công suất đường, trong khi tỷ lệ xe tải nặng và xe container tăng nhẹ trên các trục vành đai.
    \end{itemize}
    \item \textbf{Chu kỳ tuần:} Lưu lượng các ngày trong tuần (thứ Hai đến thứ Sáu) có tính ổn định cao với hệ số tương quan chéo ngày-với-ngày đạt $\rho > 0.94$. Vào thứ Bảy và Chủ Nhật, đỉnh cao điểm sáng không xuất hiện, thay vào đó là sự gia tăng lưu lượng đồng đều từ 10:00 đến 21:00 tại các trục đường thương mại và giải trí.
    \item \textbf{Biến động thời tiết mùa mưa:} Khoảng thời gian thu thập từ tháng 6 đến tháng 9 trùng với mùa mưa Nam Bộ. \end{itemize}
