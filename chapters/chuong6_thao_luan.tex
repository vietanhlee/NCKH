\chapter{Thảo luận, mô hình triển khai và hướng phát triển}
\label{chap:thao_luan}

Chương này thảo luận toàn diện về các khía cạnh kỹ thuật, tính kinh tế phần cứng và khả năng ứng dụng thực tiễn của công trình nghiên cứu trong các hệ thống giao thông thông minh (ITS) đô thị. Trọng tâm của chương là đề xuất kiến trúc triển khai phân tán biên -- đám mây (Edge--Cloud), đánh giá tính khách quan trong thực nghiệm đối sánh, phân tích khả năng chịu lỗi của hệ thống và xác định các hướng nghiên cứu mở rộng trong tương lai.

\section{Đề xuất kiến trúc triển khai phân tán biên -- đám mây}
\label{sec:kien_truc_trien_khai_edge_cloud}

Một ưu thế ứng dụng nổi bật của mô hình hai giai đoạn là khả năng phân rã bài toán và triển khai linh hoạt theo kiến trúc phân tán giữa thiết bị biên và máy chủ trung tâm:
\begin{itemize}
    \item \textbf{Tại thiết bị biên (Edge Devices -- các trạm camera ngã tư):} Mô hình Giai đoạn 1 (EfficientNet-B5 hoặc MobileNetV3 được lượng tử hóa số nguyên INT8 hoặc FP16 qua runtime nhúng như TensorRT, OpenVINO) được nạp trực tiếp vào bộ nhớ thiết bị biên. Thiết bị xử lý ảnh tĩnh tại chỗ trong bộ nhớ đệm và chỉ truyền đi một gói tin nhị phân chứa vector số lượng phương tiện: $[\hat{c}_{\text{car}}, \hat{c}_{\text{bike}}]^T$ với kích thước đúng 8 bytes.
    \item \textbf{Tại máy chủ trung tâm (Cloud Server -- trung tâm chỉ huy ITS):} Tiếp nhận các vector số đếm từ 608 nút camera qua giao thức mạng nhẹ, tổng hợp thành ma trận trạng thái mạng lưới và thực thi mô hình TA-STGCN để dự báo lưu lượng toàn đô thị trong 30 phút tới.
\end{itemize}

\subsection{Phân tích tối ưu hóa tải trọng truyền thông hạ tầng}
Trong giải pháp thu thập tập trung truyền thống, mỗi camera phải nén và truyền luồng video hoặc ảnh tĩnh độ phân giải cao ($1920 \times 1080$) về máy chủ trung tâm qua mạng cáp quang hoặc mạng di động 4G. Giả sử ảnh nén JPEG có dung lượng trung bình $500\text{ KB}$, một mạng lưới gồm 608 camera lấy mẫu định kỳ 1 phút/lần sẽ tiêu tốn:
\begin{align}
    \text{Băng thông / phút} &= 608 \times 500\text{ KB} = 304.000\text{ KB} \approx 304\text{ MB/phút} \\
    \text{Băng thông / ngày} &= 304\text{ MB} \times 60 \times 24 \approx 437.76\text{ GB/ngày}
\end{align}
Lượng dữ liệu gần nửa Terabyte mỗi ngày đặt ra gánh nặng chi phí đường truyền viễn thông rất lớn đối với ngân sách đô thị và dễ gây nghẽn mạng vào các khung giờ cao điểm hoặc khi thời tiết mưa bão.

Khi chuyển dịch việc nhận dạng thị giác xuống thiết bị biên tại từng camera, mỗi nút chỉ truyền đi một payload nhị phân gồm hai số thực 32-bit: $[\hat{c}_{\text{car}}, \hat{c}_{\text{bike}}]^T$ (đúng 8 bytes). Băng thông tiêu thụ trên toàn mạng lưới giảm xuống mức rất nhỏ:
\begin{align}
    \text{Băng thông / phút} &= 608 \times 8\text{ Bytes} \approx 4.86\text{ KB/phút} \\
    \text{Băng thông / ngày} &= 4.86\text{ KB} \times 60 \times 24 \approx 7.00\text{ MB/ngày}
\end{align}
Về mặt lý thuyết tải trọng (payload), mức độ cắt giảm kích thước dữ liệu thuần túy giữa một khung ảnh JPEG tĩnh ($500\text{ KB}$) và vector đặc trưng 2 chiều ($8\text{ Bytes}$) đạt xấp xỉ:
\begin{equation}
    \text{Tỷ lệ thu gọn payload} = \frac{500\text{ KB}}{8\text{ Bytes}} = 62.500 \text{ lần}
\end{equation}
Trong thực tế triển khai, khi tính thêm phần overhead của các tiêu đề giao thức truyền thông mạng (gói tin IP, TCP và khung tin MQTT thường chiếm khoảng vài chục đến hàng trăm bytes mỗi chu kỳ), tải trọng đường truyền thực tế vẫn giảm từ hàng trăm đến hàng nghìn lần so với phương án truyền tải hình ảnh hoặc video thô. Nhờ vậy, giải pháp giảm đáng kể áp lực lên hạ tầng viễn thông công cộng và bảo đảm khả năng vận hành ổn định ngay cả trong điều kiện mạng di động có băng thông hạn chế.

\subsection{Bảo vệ quyền riêng tư hình ảnh theo thiết kế}
Việc truyền và lưu trữ tập trung hình ảnh camera nơi công cộng tiềm ẩn nguy cơ xâm phạm quyền riêng tư của người dân (như nhận diện biển số xe cá nhân, nhận diện khuôn mặt người tham gia giao thông) và rủi ro rò rỉ dữ liệu máy chủ.

Trong kiến trúc Edge--Cloud được đề xuất, giải pháp hướng đến việc áp dụng nguyên lý bảo vệ quyền riêng tư theo thiết kế (Privacy-by-Design):
\begin{itemize}
    \item Hình ảnh camera được giải mã trực tiếp trong bộ nhớ RAM tạm thời của vi xử lý biên, đưa qua mạng nơ-ron để trích xuất số lượng xe và được giải phóng ngay lập tức.
    \item Hình ảnh thô không bị truyền qua môi trường mạng công cộng hoặc lưu trữ dài hạn tại máy chủ trung tâm, giúp hạn chế tối đa nguy cơ rò rỉ dữ liệu hình ảnh cá nhân.
    \item Dữ liệu rời khỏi camera chỉ là các con số thống kê vô danh (ví dụ: nút 102 có 12 ô tô, 45 xe máy), tuân thủ chặt chẽ các quy định pháp luật về bảo vệ dữ liệu cá nhân theo Nghị định 13/2023/NĐ-CP của Chính phủ.
\end{itemize}

\textbf{Phân tích đặc tính tài nguyên phần cứng:}
\begin{enumerate}
    \item \textbf{Vấn đề bùng nổ bộ nhớ kích hoạt của Attention đồng thời:} Mặc dù STAEformer \cite{staeformer} có số lượng tham số vừa phải (568K), cơ chế chú ý không -- thời gian kết hợp (Joint Spatio-Temporal Attention) buộc mô hình phải duy trì các tensor kích hoạt rất lớn trong VRAM, đẩy mức tiêu thụ đỉnh lên tới \textbf{11.509 MB} (chạm sát ngưỡng 12 GB của GPU phổ thông).
    \item \textbf{Đặc điểm của mô hình iTransformer:} iTransformer \cite{itransformer} tiêu tốn bộ nhớ rất thấp (265 MB) do đảo ngược chiều biến số, xem mỗi nút như một kênh độc lập mà không sử dụng ma trận không gian. Tuy nhiên, việc bỏ qua hoàn toàn thông tin hình học của mạng lưới OpenStreetMap khiến mô hình này có sai số dài hạn ở bước $t+6$ cao hơn TA-STGCN.
    \item \textbf{Cân bằng tối ưu của TA-STGCN:} Bằng cách kết hợp tích chập phổ Chebyshev $K=3$ \cite{chebnet} trên đồ thị thưa và Attention thời gian độc lập cho từng nút, TA-STGCN chỉ tiêu tốn \textbf{8.607 MB} VRAM (tiết kiệm $25.2\%$ so với STAEformer \cite{staeformer}) và giảm $70.4\%$ số lượng tham số so với ASTGCN \cite{astgcn} (453K so với 1.53M). Cấu hình này giúp một máy chủ sử dụng một GPU tầm trung thương mại (như RTX 4070/4080) có thể đảm nhiệm việc dự báo lưu lượng toàn thành phố một cách ổn định.
\end{enumerate}

\section{Tính khách quan và công bằng trong so sánh thực nghiệm}
\label{sec:tinh_cong_bang_thuc_nghiem}

Nghiên cứu tuân thủ các nguyên tắc liêm chính học thuật nhằm bảo đảm tính khách quan và minh bạch của các kết luận đối sánh:
\begin{enumerate}
    \item \textbf{Sử dụng chung hàm mục tiêu (Unified Huber Loss):} Tất cả 8 mô hình đồ thị ở Giai đoạn 2 đều được huấn luyện dưới cùng một hàm mất mát Smooth $L_1$ Huber Loss ($\delta=1.0$). Một số mô hình gốc (như Graph WaveNet \cite{wavenet}) ban đầu sử dụng $L_1$ Loss. Việc huấn luyện lại toàn bộ baseline dưới hàm Huber Loss bảo đảm đối xử đồng nhất với các điểm ngoại lai, cô lập hoàn toàn yếu tố sai khác về mặt kiến trúc.
    \item \textbf{Không tinh chỉnh tham số mang tính thiên vị:} Toàn bộ các mô hình đối chứng đều sử dụng cấu hình siêu tham số chuẩn theo công bố của các tác giả gốc (được điều chỉnh kích thước đầu vào và đầu ra cho phù hợp với đồ thị 608 nút). Nhóm nghiên cứu không thay đổi siêu tham số theo hướng làm giảm hiệu năng của các mô hình đối chứng.
    \item \textbf{Báo cáo trung thực và minh bạch kết quả:} Báo cáo nêu rõ rằng trên tập dữ liệu sạch, các mô hình đạt độ chính xác gần như tương đương nhau ($|d| < 0.1$). Đề tài không tuyên bố vượt trội trên dữ liệu sạch, mà xác định đóng góp khoa học cốt lõi là \textbf{độ bền vững vượt trội trước sai số nhận dạng cảm biến}.
\end{enumerate}

\section{Khả năng tích hợp hệ thống và cơ chế chịu lỗi mạng}
\label{sec:tich_hop_va_chiu_loi}

Để có thể vận hành trong thực tế đô thị, hệ thống cần có khả năng tích hợp linh hoạt với hạ tầng công nghệ thông tin sẵn có và duy trì hoạt động khi xảy ra sự cố truyền thông.

\subsection{Kiến trúc tích hợp truyền thông chuẩn mở (MQTT và WebSockets)}
Mô hình kết nối giữa 608 nút camera biên và máy chủ trung tâm được xây dựng trên các giao thức IoT chuẩn mở:
\begin{itemize}
    \item \textbf{Giao thức MQTT (Message Queuing Telemetry Transport):} Mỗi camera biên hoạt động như một MQTT Publisher, định kỳ 5 phút gửi bản tin nhẹ chứa mã định danh, mốc thời gian và vector số đếm:
    \begin{equation}
        \texttt{\{"node\_id": 102, "timestamp": 1756801500, "car": 14, "bike": 86\}}
    \end{equation}
    Bản tin được gửi về MQTT Broker trung tâm với mức dịch vụ QoS 1 (At least once), bảo đảm dữ liệu không bị thất lạc khi sóng di động chập chờn.
    \item \textbf{Cơ sở dữ liệu chuỗi thời gian TimescaleDB:} Máy chủ trung tâm lưu trữ dòng dữ liệu vào hệ cơ sở dữ liệu tối ưu hóa cho chuỗi thời gian, tự động phân vùng theo ngày để truy vấn lịch sử 24 bước với độ trễ dưới 2 ms.
    \item \textbf{Kênh phát dự báo WebSockets:} Sau khi mô hình TA-STGCN hoàn thành lượt suy luận ($4.76$ ms/epoch), kết quả dự báo 6 bước tương lai được phát qua WebSockets đến màn hình điều hành của trung tâm chỉ huy giao thông.
\end{itemize}

\subsection{Cơ chế dự phòng và khôi phục tín hiệu khi mất kết nối}
Trong điều kiện thực tế, hiện tượng gián đoạn kết nối do thời tiết hoặc sự cố đường truyền là khó tránh khỏi. Hệ thống được trang bị cơ chế tự phục hồi hai cấp:
\begin{enumerate}
    \item \textbf{Bộ đệm ngoại tuyến tại biên (Edge Offline Spooling):} Khi mất kết nối mạng, thiết bị biên tự động lưu tạm thời các vector số đếm vào bộ nhớ flash eMMC (dung lượng bộ đệm đủ lưu trữ liên tục 30 ngày). Khi kết nối mạng được tái lập, thiết bị biên tự động gửi bù (backfilling) để lấp đầy các khoảng trống dữ liệu trong quá khứ.
    \item \textbf{Cơ chế điền khuyết không gian bằng đồ thị (Graph Spatial Imputation):} Nếu một camera bị mất tín hiệu hoàn toàn trong thời gian thực, mô hình TA-STGCN không làm gián đoạn việc dự báo toàn mạng. Nhờ toán tử tích chập ChebNet bậc $K=3$ \cite{chebnet}, trạng thái của nút bị khuyết được xấp xỉ tự động thông qua tổ hợp tuyến tính dòng xe từ các nút lân cận trên đồ thị OpenStreetMap. Kết quả thử nghiệm cho thấy khi ngắt tín hiệu ngẫu nhiên $10\%$ số camera, sai số dự báo toàn mạng chỉ tăng nhẹ $0.18$ xe MAE, thể hiện khả năng chịu lỗi tốt của mô hình.
\end{enumerate}

\section{Các hạn chế của đề tài và hướng phát triển tiếp nối}
\label{sec:han_che_va_huong_phat_trien}

Mặc dù đã đạt được các kết quả thực nghiệm rõ ràng, nghiên cứu vẫn còn một số điểm cần tiếp tục hoàn thiện trong các giai đoạn tiếp theo:
\begin{enumerate}
    \item \textbf{Quy mô bộ dữ liệu ảnh gán nhãn:} Tập dữ liệu 5.012 ảnh bước đầu đã đáp ứng mục tiêu huấn luyện, song vẫn cần tiếp tục mở rộng để bao quát thêm nhiều góc nhìn và điều kiện thời tiết cực đoan (như mưa lớn làm mờ kính hoặc ban đêm thiếu sáng nghiêm trọng). Hướng mở rộng là áp dụng kỹ thuật học bán giám sát (Semi-Supervised Learning) hoặc các mô hình nền tảng thị giác (Vision Foundation Models) để tự động sinh giả nhãn (pseudo-labeling) trên nguồn ảnh lớn chưa gán nhãn.
    \item \textbf{Ma trận kề RBF tĩnh:} Đồ thị RBF xây dựng từ OpenStreetMap thể hiện khoảng cách vật lý cố định, chưa phản ánh tức thời các sự cố bất thường như ngập nước cục bộ, tai nạn giao thông hoặc rào chắn thi công đào đường. Hướng phát triển tiếp theo là kết hợp ma trận kề tĩnh OSM với đồ thị thích ứng tự học (Dynamic Adaptive Graph) có điều kiện biên.
    \item \textbf{Đo kiểm trên phần cứng nhúng thực tế:} Các đo đạc hiện tại được thực hiện trên GPU máy trạm. Hướng phát triển là lượng tử hóa mô hình qua TensorRT và đo kiểm thực tế về độ trễ suy luận, tốc độ khung hình (FPS) và mức tiêu thụ điện năng trên các bo mạch nhúng chuyên dụng như NVIDIA Jetson Orin Nano, Raspberry Pi 5 và Google Coral TPU.
    \item \textbf{Tích hợp các nguồn dữ liệu ngoại sinh:} Dòng giao thông đô thị chịu tác động đáng kể của các yếu tố bên ngoài như lượng mưa và các sự kiện tập trung đông người. Việc tích hợp thêm dữ liệu thời tiết và lịch sự kiện đô thị sẽ giúp nâng cao năng lực dự báo trong các tình huống đột biến.
\end{enumerate}
