\chapter{Phương pháp nghiên cứu đề xuất}
\label{chap:phuong_phap}

Chương này trình bày chi tiết kiến trúc kỹ thuật và thuật toán của khung làm việc hai giai đoạn kết hợp giữa thị giác máy tính và học sâu trên đồ thị không -- thời gian. Trọng tâm của chương là cấu trúc mô hình hồi quy đếm xe trực tiếp ở Giai đoạn 1 và mô hình mạng nơ-ron đồ thị cải tiến \textbf{TA-STGCN} ở Giai đoạn 2.

\section{Kiến trúc tổng thể khung làm việc hai giai đoạn}
\label{sec:tong_the_pipeline}

Mô hình tổng thể được tổ chức thành hai giai đoạn phân tách rõ ràng về mặt chức năng và không gian tính toán:
\begin{itemize}
    \item \textbf{Giai đoạn 1 -- Thu nhận thị giác tại camera:} Tiếp nhận ảnh tĩnh độc lập từ camera giám sát tại từng ngã tư, áp dụng mạng nơ-ron tích chập sâu với đầu hồi quy MLP đa tầng để ước lượng trực tiếp số lượng xe ô tô con và số lượng xe máy tức thời.
    \item \textbf{Giai đoạn 2 -- Dự báo đồ thị không -- thời gian toàn mạng lưới:} Tổng hợp các vector số đếm từ toàn bộ 608 camera thành chuỗi thời gian phân tán trên đồ thị không gian có hướng, áp dụng mô hình TA-STGCN để dự báo lưu lượng đa bước trong khoảng thời gian nhìn trước 30 phút ($H=6$ bước, chu kỳ 5 phút).
\end{itemize}

\section{Giai đoạn 1: Hồi quy đếm xe trực tiếp theo phân lớp}
\label{sec:giai_doan_1_hoi_quy}

\subsection{Mạng xương sống và trích xuất đặc trưng}
Mô hình tiếp nhận ảnh tĩnh đầu vào $\mathbf{I} \in \mathbb{R}^{3 \times 224 \times 224}$ đã được chuẩn hóa theo kỳ vọng và độ lệch chuẩn của tập dữ liệu ImageNet. 

Nghiên cứu lựa chọn kiến trúc \textbf{EfficientNet-B5} \cite{efficientnet} làm mạng xương sống chính thức (dựa trên kết quả tối ưu hóa đa mục tiêu về độ chính xác và lượng tính toán FLOPs được chứng minh tại Chương 5). EfficientNet-B5 bao gồm 7 cụm khối tích chập MBConv kết hợp cơ chế chú ý kênh Squeeze-and-Excitation (SE). Đầu ra của tầng tích chập cuối cùng là tensor đặc trưng bậc 3:
\begin{equation}
    \mathbf{F}_{\text{conv}} \in \mathbb{R}^{2048 \times 7 \times 7}
\end{equation}
Áp dụng tầng gộp trung bình toàn cục (Global Average Pooling -- GAP) trên hai chiều không gian:
\begin{equation}
    \mathbf{f} = \text{GAP}(\mathbf{F}_{\text{conv}}) = \frac{1}{7 \times 7} \sum_{i=1}^{7} \sum_{j=1}^{7} \mathbf{F}_{\text{conv}}(:, i, j) \in \mathbb{R}^{2048}
\end{equation}

\subsection{Đầu hồi quy MLP đa tầng thống nhất}
Để đảm bảo tính so sánh công bằng giữa các kiến trúc backbone khác nhau, tầng phân loại 1.000 lớp ImageNet gốc được thay thế bằng cùng một đầu hồi quy MLP đa tầng thống nhất:
\begin{align}
    \mathbf{h}_1 &= \text{LayerNorm}(\mathbf{W}_1 \mathbf{f} + \mathbf{b}_1), \quad \mathbf{W}_1 \in \mathbb{R}^{128 \times 2048}, \ \mathbf{b}_1 \in \mathbb{R}^{128} \\
    \mathbf{h}_2 &= \text{ReLU}(\mathbf{h}_1) \in \mathbb{R}^{128} \\
    \hat{\mathbf{c}} &= \mathbf{W}_2 \mathbf{h}_2 + \mathbf{b}_2, \quad \mathbf{W}_2 \in \mathbb{R}^{2 \times 128}, \ \mathbf{b}_2 \in \mathbb{R}^{2}
\end{align}
trong đó $\hat{\mathbf{c}} = [\hat{c}_{\text{car}}, \hat{c}_{\text{bike}}]^T \in \mathbb{R}^2$ là vector số lượng xe ô tô con và xe máy dự đoán.

\textbf{Đặc điểm thiết kế của đầu hồi quy:}
\begin{itemize}
    \item Tầng ẩn 128 chiều đóng vai trò như một phễu nén đặc trưng, buộc mô hình phải chắt lọc các biểu diễn cô đọng nhất liên quan đến số lượng xe thay vì học thuộc lòng cảnh nền.
    \item Tầng LayerNorm được đặt trước hàm kích hoạt ReLU giúp ổn định phân phối giá trị kích hoạt, ngăn ngừa hiện tượng suy thoái nơ-ron khi huấn luyện trên dữ liệu ảnh có độ biến động ánh sáng lớn.
\end{itemize}

\subsection{Hàm mất mát Smooth L1 (Huber Loss)}
Để huấn luyện mô hình Giai đoạn 1 có khả năng chống chịu bền bỉ trước các ảnh bị lóa sáng hoặc che khuất đột ngột, hàm mất mát Huber Loss \cite{huber} ($\delta = 1.0$) được áp dụng cho từng phân lớp:
\begin{align}
    e_{\text{car}} &= c_{\text{car}} - \hat{c}_{\text{car}} \\
    e_{\text{bike}} &= c_{\text{bike}} - \hat{c}_{\text{bike}}
\end{align}
Hàm mục tiêu tối ưu hóa toàn cục của Giai đoạn 1 là:
\begin{equation}
    \mathcal{L}_{\text{Stage 1}} = \frac{1}{2} \left( \mathcal{L}_{\mathrm{Huber}}(e_{\text{car}}) + \mathcal{L}_{\mathrm{Huber}}(e_{\text{bike}}) \right)
\end{equation}
với:
\begin{equation}
\mathcal{L}_{\mathrm{Huber}}(e) =
\begin{cases}
\frac{1}{2} e^2, & \text{khi } |e| < 1.0, \\
|e| - 0.5, & \text{khi } |e| \ge 1.0.
\end{cases}
\end{equation}

\section{Giai đoạn 2: Mô hình dự báo đồ thị không -- thời gian TA-STGCN}
\label{sec:giai_doan_2_tastgcn}

\subsection{Cấu trúc tensor đặc trưng đa chiều}
Tại mỗi thời điểm dự báo $t$, đầu vào của mô hình là một chuỗi quan sát lịch sử gồm $T_{\text{in}} = 24$ bước thời gian (tương đương 120 phút quan sát). Tại mỗi bước thời gian $\tau \in \{t - T_{\text{in}} + 1, \dots, t\}$ và tại mỗi nút camera $i \in \{1, \dots, N\}$, một vector đặc trưng $F = 5$ chiều được cấu tạo:
\begin{enumerate}
    \item Số lượng xe ô tô con đã chuẩn hóa z-score theo từng nút: $\tilde{c}_{\text{car}}^{(i, \tau)}$.
    \item Số lượng xe máy đã chuẩn hóa z-score theo từng nút: $\tilde{c}_{\text{bike}}^{(i, \tau)}$.
    \item Đặc trưng sin của chu kỳ ngày đêm: $\sin\left(\frac{2\pi \cdot \text{tod}(\tau)}{1440}\right)$.
    \item Đặc trưng cos của chu kỳ ngày đêm: $\cos\left(\frac{2\pi \cdot \text{tod}(\tau)}{1440}\right)$.
    \item Chỉ số giờ chuẩn hóa trong ngày: $\frac{\text{hour}(\tau)}{24}$.
\end{enumerate}
với $\text{tod}(\tau)$ là số phút tính từ đầu ngày (từ 0 đến 1439). 

Tensor đầu vào cho một mini-batch gồm $B$ mẫu có kích thước:
\begin{equation}
    \mathbf{X} \in \mathbb{R}^{B \times T_{\text{in}} \times N \times 5}
\end{equation}
Mục tiêu là dự báo ma trận lưu lượng trong $H = 6$ bước thời gian kế tiếp (30 phút tới):
\begin{equation}
    \hat{\mathbf{Y}} \in \mathbb{R}^{B \times H \times N \times 2}
\end{equation}

\subsection{Chi tiết thiết kế khối ST-Conv}
Khối ST-Conv được thiết kế theo cấu trúc sandwich gồm ba tầng tuần tự:
\begin{enumerate}
    \item \textbf{Tầng tích chập thời gian GLU thứ nhất:} Tiếp nhận tensor đặc trưng, áp dụng tích chập 1D dọc theo trục thời gian với kernel $K_t = 3$ để mở rộng số kênh ẩn lên $C = 80$:
    \begin{equation}
        \mathbf{H}_1 = (\mathbf{X} \star \boldsymbol{\Theta}_{1,1} + \mathbf{b}_{1,1}) \odot \sigma(\mathbf{X} \star \boldsymbol{\Theta}_{1,2} + \mathbf{b}_{1,2}) \in \mathbb{R}^{B \times C \times N \times T_{\text{in}}}
    \end{equation}
    \item \textbf{Tầng tích chập không gian ChebNet:} Áp dụng đa thức Chebyshev bậc $K = 3$ \cite{chebnet} trên ma trận Laplacian $\tilde{\mathbf{L}}$ của đồ thị OpenStreetMap:
    \begin{equation}
        \mathbf{H}_2 = \text{ReLU}\left( \sum_{k=0}^{K-1} T_k(\tilde{\mathbf{L}}) \mathbf{H}_1 \boldsymbol{\Theta}_k^{\text{spatial}} \right) \in \mathbb{R}^{B \times C \times N \times T_{\text{in}}}
    \end{equation}
    với $\boldsymbol{\Theta}_k^{\text{spatial}} \in \mathbb{R}^{C \times C}$ là ma trận trọng số biến đổi kênh học được.
    \item \textbf{Tầng tích chập thời gian GLU thứ hai:} Tiếp tục tinh lọc đặc trưng thời gian sau khi đã được tổng hợp không gian:
    \begin{equation}
        \mathbf{H}_3 = (\mathbf{H}_2 \star \boldsymbol{\Theta}_{2,1} + \mathbf{b}_{2,1}) \odot \sigma(\mathbf{H}_2 \star \boldsymbol{\Theta}_{2,2} + \mathbf{b}_{2,2}) \in \mathbb{R}^{B \times C \times N \times T_{\text{in}}}
    \end{equation}
    Đầu ra của khối được kết hợp với kết nối tắt Residual, Dropout ($p = 0.25$) và chuẩn hóa LayerNorm:
    \begin{equation}
        \mathbf{H}_{\text{ST-Conv}} = \text{LayerNorm}\left( \mathbf{H}_{\text{residual}} + \text{Dropout}(\mathbf{H}_3, p=0.25) \right)
    \end{equation}
\end{enumerate}

\subsection{Mô-đun chú ý thời gian đa đầu đặt ở giữa}
Điểm cải tiến kiến trúc cốt lõi của \textbf{TA-STGCN} là tích hợp mô-đun chú ý thời gian đa đầu (Multi-Head Temporal Self-Attention) vào vị trí giữa hai khối ST-Conv.

\subsubsection{Cơ chế định hình lại tensor và tính toán attention}
Gọi đầu ra sau khối ST-Conv thứ nhất là $\mathbf{H}^{(1)} \in \mathbb{R}^{B \times C \times N \times T_{\text{in}}}$ với $C = 80$ kênh và $T_{\text{in}} = 24$ bước. Để tránh chi phí tương tác bậc hai theo số nút, tensor được hoán vị và làm phẳng:
\begin{equation}
    \mathbf{Z} = \text{Reshape}(\text{Permute}(\mathbf{H}^{(1)}, [0, 2, 3, 1])) \in \mathbb{R}^{(B \cdot N) \times T_{\text{in}} \times C}
\end{equation}
Mỗi nút trong batch được xem như một chuỗi độc lập có độ dài $T_{\text{in}} = 24$. Áp dụng cơ chế Multi-Head Attention \cite{attention} với $h = 4$ đầu (chiều mỗi đầu $d_k = C/h = 20$):
\begin{align}
    \mathbf{Q}_i &= \mathbf{Z} \mathbf{W}_i^Q \in \mathbb{R}^{(B \cdot N) \times T_{\text{in}} \times d_k} \\
    \mathbf{K}_i &= \mathbf{Z} \mathbf{W}_i^K \in \mathbb{R}^{(B \cdot N) \times T_{\text{in}} \times d_k} \\
    \mathbf{V}_i &= \mathbf{Z} \mathbf{W}_i^V \in \mathbb{R}^{(B \cdot N) \times T_{\text{in}} \times d_k} \quad (\forall i = 1, \dots, h) \\
    \text{head}_i &= \text{Softmax}\left(\frac{\mathbf{Q}_i \mathbf{K}_i^T}{\sqrt{d_k}}\right) \mathbf{V}_i \in \mathbb{R}^{(B \cdot N) \times T_{\text{in}} \times d_k} \\
    \text{MultiHead}(\mathbf{Z}) &= \text{concat}\left( \text{head}_1, \text{head}_2, \dots, \text{head}_h \right) \mathbf{W}^O \in \mathbb{R}^{(B \cdot N) \times T_{\text{in}} \times C}
\end{align}

\subsubsection{Khối mạng truyền thẳng và chuẩn hóa residual}
Đầu ra sau Attention được đưa qua mạng truyền thẳng hai tầng tuyến tính với kích hoạt GELU và hệ số mở rộng kênh gấp 4 lần ($4C = 320$ chiều):
\begin{align}
    \mathbf{Z}_1 &= \text{LayerNorm}(\mathbf{Z} + \text{Dropout}(\text{MultiHead}(\mathbf{Z}), p=0.1)) \\
    \text{FFN}(\mathbf{Z}_1) &= \text{GELU}(\mathbf{Z}_1 \mathbf{W}_{\text{ffn}, 1} + \mathbf{b}_{\text{ffn}, 1}) \mathbf{W}_{\text{ffn}, 2} + \mathbf{b}_{\text{ffn}, 2} \\
    \mathbf{Z}_{\text{out}} &= \text{LayerNorm}(\mathbf{Z}_1 + \text{Dropout}(\text{FFN}(\mathbf{Z}_1), p=0.1))
\end{align}
với $\mathbf{W}_{\text{ffn}, 1} \in \mathbb{R}^{C \times 4C}$ và $\mathbf{W}_{\text{ffn}, 2} \in \mathbb{R}^{4C \times C}$.

Tensor $\mathbf{Z}_{\text{out}}$ sau đó được khôi phục kích thước $\mathbb{R}^{B \times C \times N \times T_{\text{in}}}$ và truyền vào khối ST-Conv thứ hai để tiếp tục xử lý đặc trưng trước khi xuất ra tầng dự báo.

\subsection{Tầng chiếu thời gian ngõ ra và hàm mục tiêu huấn luyện}
Sau khi đi qua khối ST-Conv thứ hai, tensor có kích thước $\mathbf{H}^{(2)} \in \mathbb{R}^{(B \cdot N) \times C \times T_{\text{in}}}$. Mô hình áp dụng một tầng tích chập 1D theo chiều thời gian với kích thước kernel bằng đúng $T_{\text{in}} = 24$ để thu gọn chiều thời gian về 1:
\begin{equation}
    \mathbf{Y}_{\text{proj}} = \text{Conv1D}(\mathbf{H}^{(2)}, \text{kernel\_size}=T_{\text{in}}) \in \mathbb{R}^{(B \cdot N) \times (H \cdot 2) \times 1}
\end{equation}
Tensor được định hình lại thành $\hat{\mathbf{Y}} \in \mathbb{R}^{B \times H \times N \times 2}$ tương ứng với dự báo 6 bước tương lai cho 2 loại xe trên toàn bộ 608 nút.

Mô hình được tối ưu hóa bằng hàm mất mát Huber Loss trên toàn bộ các nút và các bước dự báo:
\begin{equation}
    \mathcal{L}_{\text{Stage 2}} = \frac{1}{B \cdot H \cdot N \cdot 2} \sum_{b=1}^{B} \sum_{h=1}^{H} \sum_{i=1}^{N} \sum_{m=1}^{2} \mathcal{L}_{\mathrm{Huber}}\left( Y_{b,h,i,m} - \hat{Y}_{b,h,i,m} \right)
\end{equation}
với ngưỡng chuyển tiếp chuẩn $\delta = 1.0$.

\section{Thuật toán huấn luyện và suy luận toàn diện}
\label{sec:thuat_toan_huan_luyen}

Thuật toán \ref{algo:training_pipeline} mô tả tóm tắt toàn bộ quy trình huấn luyện hai giai đoạn từ ảnh thô camera đến mô hình dự báo đồ thị hoàn chỉnh.

\begin{algorithm_box}{Quy trình huấn luyện hai giai đoạn}{
\textbf{Đầu vào:} Tập ảnh camera $\mathcal{D}_{\text{vision}} = \{(I_k, \mathbf{c}_k)\}_{k=1}^{5012}$, Đồ thị đường phố OpenStreetMap $\mathcal{G}_{\text{OSM}}$, Ma trận trọng số $W$. \\
\textbf{Đầu ra:} Mô hình thị giác $f_{\theta^*}$, Mô hình đồ thị dự báo $\mathcal{M}_{\Phi^*}$. \\
\textbf{Pha 1: Huấn luyện mô hình thị giác (Giai đoạn 1):}
\begin{enumerate}
    \item Khởi tạo backbone EfficientNet-B5 với trọng số tiền huấn luyện ImageNet.
    \item Khởi tạo đầu hồi quy MLP ngẫu nhiên.
    \item \textbf{Lặp} qua các epoch từ 1 đến 120:
    \begin{itemize}
        \item Lấy batch ảnh $B=32$, áp dụng tăng cường dữ liệu bảo toàn nhãn.
        \item Tính dự đoán $\hat{\mathbf{c}} = f_\theta(I)$ và tính hàm mất mát Huber Loss $\mathcal{L}_{\text{Stage 1}}$.
        \item Cập nhật trọng số bằng thuật toán tối ưu AdamW \cite{adam} kết hợp Cosine Annealing.
    \end{itemize}
    \item Lưu trọng số tối ưu $\theta^*$ theo sai số MAE nhỏ nhất trên tập Validation.
\end{enumerate}
\textbf{Pha 2: Sinh dữ liệu chuỗi thời gian và tiền xử lý:}
\begin{enumerate}
    \item Sử dụng $f_{\theta^*}$ suy luận liên tục trên 608 camera trong 86 ngày để tạo chuỗi 1 phút.
    \item Áp dụng bộ lọc trượt kép nhân quả và lấy mẫu đều đặn $\Delta t = 5$ phút để tạo tensor $\mathbf{Y}_{\text{total}} \in \mathbb{R}^{24768 \times 608 \times 2}$.
    \item Phân chia chuỗi thời gian: $80\%$ Train, $10\%$ Validation, $10\%$ Test.
    \item Chuẩn hóa z-score theo từng nút và đóng băng thông số $\mu_i, \sigma_i$ từ tập Train.
\end{enumerate}
\textbf{Pha 3: Huấn luyện mô hình đồ thị TA-STGCN (Giai đoạn 2):}
\begin{enumerate}
    \item Tính toán ma trận Laplacian chuẩn hóa $\tilde{\mathbf{L}}$ từ ma trận khoảng cách RBF.
    \item Khởi tạo mô hình TA-STGCN với 2 khối ST-Conv và Temporal Multi-Head Attention ($h=4$).
    \item \textbf{Lặp} qua các epoch từ 1 đến 120:
    \begin{itemize}
        \item Lấy batch chuỗi thời gian $B=64$, cửa sổ $T_{\text{in}}=24$, mục tiêu $H=6$.
        \item Lan truyền xuôi qua ST-Conv 1 $\to$ Temporal Attention $\to$ ST-Conv 2 $\to$ Conv1D Output.
        \item Tính hàm mất mát Huber Loss $\mathcal{L}_{\text{Stage 2}}$ và cập nhật trọng số bằng thuật toán Adam \cite{adam}.
        \item Điều chỉnh tốc độ học theo cơ chế ReduceLROnPlateau.
    \end{itemize}
    \item Dừng sớm sau 20 epoch nếu Validation MAE không cải thiện. Lưu trọng số tối ưu $\Phi^*$.
\end{enumerate}
}
\label{algo:training_pipeline}
\end{algorithm_box}

\textbf{Lý do thiết kế vị trí đặt của Temporal Attention ở giữa:}
Việc đặt tầng Multi-Head Temporal Self-Attention ở giữa hai khối ST-Conv mang lại lợi thế cấu trúc rõ nét:
\begin{enumerate}
    \item \textbf{Khối ST-Conv 1 đóng vai trò trích xuất đặc trưng cục bộ:} Tầng ChebNet thứ nhất đã kết hợp thông tin cấu trúc 2-hop từ các nút lân cận, giúp các vector đặc trưng đưa vào tầng Attention chứa đựng ngữ cảnh cục bộ của nút giao thay vì chỉ là tín hiệu rời rạc.
    \item \textbf{Tầng Attention thực hiện tái cân bằng trọng số thời gian toàn cục:} Tầng Attention tính toán ma trận tương quan giữa tất cả các bước thời gian còn lại ($T=20$), giúp triệt tiêu các dao động nhiễu đột ngột từ cảm biến trước khi bước vào khối ST-Conv thứ hai.
    \item \textbf{Khối ST-Conv 2 thực hiện tinh chỉnh không gian thứ cấp:} Tín hiệu thời gian đã được làm mượt tiếp tục được khuếch tán không gian qua đồ thị OSM, bảo đảm tính liên tục không -- thời gian trước khi chiếu sang các bước dự báo tương lai.
\end{enumerate}

\section{Quy trình vận hành suy luận thời gian thực}
\label{sec:quy_trinh_suy_luan}

Để ứng dụng mô hình vào trung tâm điều hành giao thông đô thị thực tế, quy trình suy luận được tổ chức thành 5 bước liên hoàn:
\begin{enumerate}
    \item \textbf{Bước 1 -- Thu nhận và xử lý ảnh tại biên:} Cứ sau mỗi chu kỳ $\Delta t = 5$ phút, camera tại 608 ngã tư chụp ảnh hiện trường. Mạng EfficientNet-B5 lượng tử hóa INT8 thực thi suy luận trong khoảng $35$ ms, trích xuất vector số lượng xe tức thời $[\hat{c}_{\text{car}}, \hat{c}_{\text{bike}}]^T$.
    \item \textbf{Bước 2 -- Cập nhật bộ đệm chuỗi thời gian:} Máy chủ trung tâm nhận 608 vector số đếm (tổng cộng $4.86\text{ KB}$), đẩy vào hàng đệm trượt FIFO có độ dài cố định $T_{\text{in}} = 24$ bước (120 phút gần nhất). Điểm dữ liệu cũ nhất tại bước $t-24$ bị loại bỏ.
    \item \textbf{Bước 3 -- Chuẩn hóa z-score cục bộ:} Dữ liệu trong hàng đệm được chuẩn hóa bằng tham số $\mu_i, \sigma_i$ cố định tương ứng của từng nút: $x_{i, t} = (y_{i, t} - \mu_i) / \sigma_i$.
    \item \textbf{Bước 4 -- Dự báo toàn mạng qua TA-STGCN:} Tensor kích thước $(1, 2, 24, 608)$ được đưa qua mô hình TA-STGCN trên GPU máy chủ trung tâm. Thời gian thực thi suy luận toàn mạng chỉ mất đúng \textbf{$4.76$ ms}.
    \item \textbf{Bước 5 -- Giải chuẩn hóa và phát tín hiệu điều phối:} Đầu ra được giải chuẩn hóa về số lượng xe thực tế: $\hat{y}_{i, t+h} = \hat{z}_{i, t+h} \cdot \sigma_i + \mu_i$. Các vector dự báo 6 bước tương lai ($t+5$ đến $t+30$ phút) được truyền qua API đến bộ điều khiển tín hiệu đèn giao thông và tích hợp lên hệ thống thông tin giám sát đô thị.
\end{enumerate}
