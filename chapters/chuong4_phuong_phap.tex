\chapter{Thiết kế hệ thống Two-Stage Pipeline}
\label{chap:phuong_phap}

Chương này trình bày chi tiết thiết kế kỹ thuật, đặc tả các giao diện dữ liệu vào/ra và thuật toán vận hành của hệ thống \textbf{Two-Stage Pipeline}. Trên cơ sở các nền tảng toán học và mô hình lý thuyết đã được làm sáng tỏ tại Chương \ref{chap:co_so_ly_thuyet}, thiết kế hệ thống tập trung vào việc hiện thực hóa hai phân hệ xử lý nối tầng: Phân hệ ước lượng lưu lượng phương tiện từ ảnh thị giác (Giai đoạn 1) và Phân hệ dự báo dòng giao thông không -- thời gian bằng mạng đồ thị cải tiến \textbf{TA-STGCN} (Giai đoạn 2).

\section{Kiến trúc tổng thể hệ thống Two-Stage Pipeline}
\label{sec:tong_the_pipeline}

Hệ thống được thiết kế theo kiến trúc nối tầng phân tách độc lập (Decoupled Pipeline Architecture), phân rã bài toán lớn thành hai phân hệ chuyên biệt về mặt chức năng và môi trường tính toán:
\begin{itemize}
    \item \textbf{Phân hệ 1 -- Cảm nhận thị giác tại camera tuyến đường:} Tiếp nhận các khung hình ảnh tĩnh độc lập thu nhận từ hệ thống camera giám sát trên từng tuyến đường, áp dụng mạng nơ-ron tích chập sâu kết hợp đầu hồi quy MLP đa tầng để ước lượng trực tiếp số lượng xe ô tô con và số lượng xe máy tức thời.
    \item \textbf{Phân hệ 2 -- Dự báo đồ thị không -- thời gian toàn mạng lưới:} Tổng hợp các vector số đếm từ toàn bộ 608 nút camera trên các tuyến đường thành chuỗi thời gian phân tán trên đồ thị không gian có hướng, áp dụng mô hình mạng đồ thị TA-STGCN để dự báo lưu lượng đa bước trong khoảng thời gian nhìn trước 30 phút ($H=6$ bước, chu kỳ 5 phút).
\end{itemize}

\textbf{Nguyên tắc thiết kế phân tách độc lập (Decoupling Principle):}
Hai phân hệ được liên kết lỏng (loosely coupled) thông qua một giao diện dữ liệu tiêu chuẩn là vector số lượng phương tiện theo phân lớp $\hat{\mathbf{c}} = [\hat{c}_{\text{car}}, \hat{c}_{\text{bike}}]^T \in \mathbb{R}^2$. Cách tiếp cận này mang lại ba ưu thế kỹ thuật cốt lõi:
\begin{enumerate}
    \item \textbf{Tối ưu hóa tài nguyên tính toán độc lập:} Phân hệ 1 xử lý cục bộ trên từng khung ảnh tĩnh tại thiết bị biên (Edge) mà không cần chia sẻ thông tin giữa các camera lân cận; trong khi Phân hệ 2 tập trung xử lý chuỗi thời gian và tương quan không gian mạng lưới trên máy chủ trung tâm (Cloud).
    \item \textbf{Bảo toàn tính mở của hệ thống:} Cho phép nâng cấp hoặc thay thế độc lập kiến trúc mạng thị giác (như chuyển đổi giữa EfficientNet-B5 và MobileNetV3) mà không ảnh hưởng đến cấu trúc mạng nơ-ron đồ thị ở Giai đoạn 2.
    \item \textbf{Bảo mật và tối ưu truyền dẫn:} Loại bỏ hoàn toàn nhu cầu truyền luồng video dung lượng lớn qua mạng Internet công cộng, chỉ truyền tải dữ liệu số đếm dạng số thực nhỏ gọn.
\end{enumerate}

\section{Thiết kế Phân hệ 1: Ước lượng lưu lượng phương tiện từ ảnh camera}
\label{sec:giai_doan_1_hoi_quy}

\subsection{Đặc tả giao diện dữ liệu vào và ra}
Phân hệ 1 được thiết kế với giao diện dữ liệu chuẩn mực nhằm phục vụ việc tích hợp vào các thiết bị camera giám sát giao thông thực tế:
\begin{itemize}
    \item \textbf{Đầu vào (Input):} Ảnh tĩnh màu RGB $\mathbf{I} \in \mathbb{R}^{3 \times 224 \times 224}$ thu nhận từ camera giao thông trên các tuyến đường quan sát, được chuẩn hóa theo kỳ vọng và độ lệch chuẩn của tập dữ liệu ImageNet:
    \begin{equation}
        \mathbf{I}_{\text{norm}} = \frac{\mathbf{I} - \boldsymbol{\mu}_{\text{ImageNet}}}{\boldsymbol{\sigma}_{\text{ImageNet}}}
    \end{equation}
    \item \textbf{Đầu ra (Output):} Vector số lượng phương tiện tức thời $\hat{\mathbf{c}} = [\hat{c}_{\text{car}}, \hat{c}_{\text{bike}}]^T \in \mathbb{R}^2_{\ge 0}$, tương ứng với số xe ô tô con và số xe máy xuất hiện trong khung hình quan sát.
\end{itemize}

\subsection{Cấu hình mạng xương sống và trích xuất đặc trưng}
Kế thừa các phân tích lý thuyết về cơ chế mở rộng đồng thời (Compound Scaling) tại Mục \ref{sec:kien_truc_efficientnet}, hệ thống lựa chọn mạng xương sống \textbf{EfficientNet-B5} để trích xuất đặc trưng hình ảnh. Mạng gồm 7 cụm khối tích chập MBConv kết hợp cơ chế chú ý kênh Squeeze-and-Excitation (SE).

Tensor đặc trưng đầu ra tại tầng tích chập cuối cùng có kích thước:
\begin{equation}
    \mathbf{F}_{\text{conv}} \in \mathbb{R}^{2048 \times 7 \times 7}
\end{equation}
Để nén chiều không gian và tạo tính bất biến trước các dịch chuyển nhỏ của phương tiện trên mặt đường, tầng gộp trung bình toàn cục (Global Average Pooling -- GAP) được áp dụng:
\begin{equation}
    \mathbf{f} = \text{GAP}(\mathbf{F}_{\text{conv}}) \in \mathbb{R}^{2048}
\end{equation}

\subsection{Đặc tả đầu hồi quy MLP đa tầng thống nhất}
Vector đặc trưng $\mathbf{f}$ được đưa qua đầu hồi quy nén (Bottleneck MLP Head) nhằm ánh xạ trực tiếp sang vector số lượng phương tiện:
\begin{align}
    \mathbf{h}_1 &= \text{LayerNorm}(\mathbf{W}_1 \mathbf{f} + \mathbf{b}_1), \quad \mathbf{W}_1 \in \mathbb{R}^{128 \times 2048}, \ \mathbf{b}_1 \in \mathbb{R}^{128} \\
    \mathbf{h}_2 &= \text{ReLU}(\mathbf{h}_1) \in \mathbb{R}^{128} \\
    \hat{\mathbf{c}} &= \mathbf{W}_2 \mathbf{h}_2 + \mathbf{b}_2, \quad \mathbf{W}_2 \in \mathbb{R}^{2 \times 128}, \ \mathbf{b}_2 \in \mathbb{R}^{2}
\end{align}

\subsection{Hàm tổn thất huấn luyện}
Phân hệ 1 được tối ưu hóa bằng hàm tổn thất Huber Loss ($\delta = 1.0$) cho từng phân loại phương tiện (theo nguyên lý bền vững tại Mục \ref{sec:huber_loss}):
\begin{equation}
    \mathcal{L}_{\text{Stage 1}} = \frac{1}{2} \left( \mathcal{L}_{\mathrm{Huber}}(c_{\text{car}} - \hat{c}_{\text{car}}) + \mathcal{L}_{\mathrm{Huber}}(c_{\text{bike}} - \hat{c}_{\text{bike}}) \right)
\end{equation}
Thiết kế này bảo đảm mô hình huấn luyện ổn định, hạn chế ảnh hưởng của các mẫu ngoại lai xuất hiện khi mặt đường bị lóa đèn pha hoặc ngập nước.

\section{Thiết kế Phân hệ 2: Dự báo đồ thị không -- thời gian TA-STGCN}
\label{sec:giai_doan_2_tastgcn}

\subsection{Đặc tả giao diện dữ liệu vào và ra}
Phân hệ 2 tiếp nhận dữ liệu chuỗi thời gian phân tán từ toàn bộ các tuyến đường và thực hiện suy luận dự báo đồng thời trên toàn mạng lưới:
\begin{itemize}
    \item \textbf{Đầu vào (Input):} 
    \begin{enumerate}
        \item Tensor lịch sử quan sát $\mathbf{X} \in \mathbb{R}^{B \times T_{\text{in}} \times N \times F}$, trong đó $B$ là kích thước mini-batch, $N = 608$ nút camera trên các tuyến đường, $T_{\text{in}} = 24$ bước quan sát lịch sử (chu kỳ 5 phút/bước, tương đương 120 phút), và $F = 5$ đặc trưng gồm: lưu lượng xe ô tô con đã chuẩn hóa z-score, lưu lượng xe máy đã chuẩn hóa z-score, đặc trưng $\sin$ và $\cos$ của thời điểm trong ngày (Time-of-Day), cùng chỉ số giờ chuẩn hóa.
        \item Ma trận kề trọng số $\mathbf{W} \in \mathbb{R}^{N \times N}$ phản ánh cấu trúc liên thông và khoảng cách vật lý của mạng lưới đường bộ OpenStreetMap.
    \end{enumerate}
    \item \textbf{Đầu ra (Output):} Tensor lưu lượng dự báo đa bước tương lai:
    \begin{equation}
        \hat{\mathbf{Y}} \in \mathbb{R}^{B \times H \times N \times 2}
    \end{equation}
    cho $H = 6$ bước thời gian kế tiếp ($t+5, t+10, \dots, t+30$ phút) trên toàn bộ $N = 608$ nút cho cả 2 loại phương tiện (ô tô con và xe máy).
\end{itemize}

\subsection{Cấu trúc tích hợp mô-đun trong kiến trúc TA-STGCN}
Mô hình TA-STGCN được cấu thành từ hai khối tích chập không -- thời gian ST-Conv dạng sandwich kết hợp với mô-đun Self-Attention theo thời gian đa đầu (Multi-Head Temporal Self-Attention) được bố trí ở giữa:

\begin{enumerate}
    \item \textbf{Khối ST-Conv thứ nhất (Trích xuất đặc trưng cục bộ):}
    \begin{itemize}
        \item Tầng tích chập thời gian 1D có cổng GLU thứ nhất (kernel thời gian $K_t = 3$, đệm nhân quả $\text{pad}=1$ bảo toàn độ dài $T_{\text{in}}=24$, số kênh ẩn $C = 80$) trích xuất tương quan thời gian ngắn hạn.
        \item Tầng tích chập đồ thị phổ ChebNet (đa thức Chebyshev bậc $K = 3$ trên ma trận Laplacian $\tilde{\mathbf{L}}$ của đồ thị OpenStreetMap) tổng hợp đặc trưng không gian từ các nút lân cận trong phạm vi 2-hop.
        \item Tầng tích chập thời gian GLU thứ hai tiếp tục tinh chế đặc trưng, kết hợp với nhánh tắt Residual, Dropout ($p=0.25$) và chuẩn hóa LayerNorm trên từng nút.
    \end{itemize}

    \item \textbf{Mô-đun Multi-Head Temporal Self-Attention (Lọc nhiễu thích ứng):}
    Tensor sau khối ST-Conv 1 có kích thước $\mathbf{H}^{(1)} \in \mathbb{R}^{(B \cdot N) \times T_{\text{in}} \times C}$ (với $T_{\text{in}}=24, C=80$). Mô-đun Attention gồm $h = 4$ đầu độc lập (kích thước mỗi đầu $d_k = C/h = 20$), tính toán ma trận tự chú ý dọc theo trục thời gian cho từng nút mạng:
    \begin{equation}
        \text{head}_i = \text{Softmax}\left(\frac{\mathbf{Q}_i \mathbf{K}_i^T}{\sqrt{d_k}}\right) \mathbf{V}_i \in \mathbb{R}^{(B \cdot N) \times T_{\text{in}} \times d_k}
    \end{equation}
    \textbf{Vai trò thiết kế cốt lõi:} Mô-đun Attention đóng vai trò như một bộ lọc thông thấp thích ứng. Khi một camera tại tuyến đường gặp sự cố thị giác (như lóa đèn hoặc che khuất đột ngột dẫn đến giá trị đếm bị nhảy vọt cục bộ), cơ chế chú ý sẽ phân bổ trọng số thấp cho bước thời gian dị thường này và tái cân bằng dựa trên xu hướng ổn định của chuỗi thời gian, ngăn chặn hiện tượng lan truyền sai số sang các nút lân cận.

    \item \textbf{Khối ST-Conv thứ hai (Tinh chỉnh không gian thứ cấp):}
    Tiếp nhận tensor thời gian đã được làm mượt và lọc nhiễu, tiếp tục khuếch tán không gian qua đồ thị nhằm bảo đảm tính liên tục và nhất quán của dòng giao thông trước khi đưa vào tầng xuất dự báo.

    \item \textbf{Tầng chiếu ngõ ra (Output Block):}
    Áp dụng tầng tích chập 1D theo chiều thời gian để thu gọn cửa sổ lịch sử về chiều dài 1, tiếp tục chiếu qua tầng biến đổi kênh để xuất ra đúng $H \times 2 = 12$ giá trị dự báo cho mỗi nút:
    \begin{equation}
        \hat{\mathbf{Y}} = \text{Reshape}\left( \text{Conv1D}\left(\mathbf{H}^{(2)}\right) \right) \in \mathbb{R}^{B \times H \times N \times 2}
    \end{equation}
\end{enumerate}

\subsection{Hàm mục tiêu huấn luyện toàn mạng}
Mô hình Phân hệ 2 được tối ưu hóa đồng thời trên tất cả các bước dự báo nhìn trước và toàn bộ các nút camera:
\begin{equation}
    \mathcal{L}_{\text{Stage 2}} = \frac{1}{B \cdot H \cdot N \cdot 2} \sum_{b=1}^{B} \sum_{h=1}^{H} \sum_{i=1}^{N} \sum_{m=1}^{2} \mathcal{L}_{\mathrm{Huber}}\left( Y_{b,h,i,m} - \hat{Y}_{b,h,i,m} \right)
\end{equation}
với ngưỡng chuyển đổi $\delta = 1.0$.

\section{Thuật toán huấn luyện và quy trình suy luận toàn diện}
\label{sec:thuat_toan_huan_luyen}

Thuật toán \ref{algo:training_pipeline} mô tả tóm tắt toàn bộ quy trình huấn luyện Two-Stage Pipeline từ ảnh thô camera đến mô hình dự báo đồ thị hoàn chỉnh.

\begin{algorithm_box}{Quy trình huấn luyện Two-Stage Pipeline}{
\textbf{Đầu vào:} Tập ảnh camera $\mathcal{D}_{\text{vision}} = \{(I_k, \mathbf{c}_k)\}_{k=1}^{5012}$, Đồ thị đường phố OpenStreetMap $\mathcal{G}_{\text{OSM}}$, Ma trận trọng số $W$. \\
\textbf{Đầu ra:} Mô hình thị giác $f_{\theta^*}$, Mô hình đồ thị dự báo $\mathcal{M}_{\Phi^*}$. \\
\textbf{Pha 1: Huấn luyện mô hình thị giác (Giai đoạn 1):}
\begin{enumerate}
    \item Khởi tạo backbone EfficientNet-B5 với trọng số tiền huấn luyện ImageNet.
    \item Khởi tạo đầu hồi quy MLP ngẫu nhiên.
    \item \textbf{Lặp} qua các epoch từ 1 đến 120:
    \begin{itemize}
        \item Lấy batch ảnh $B=32$, áp dụng tăng cường dữ liệu bảo toàn nhãn.
        \item Lan truyền xuôi: $\hat{\mathbf{c}} = \text{MLP}(\text{GAP}(\text{Backbone}(I)))$.
        \item Tính hàm mất mát $\mathcal{L}_{\text{Stage 1}}$ bằng Smooth L1 (Huber Loss).
        \item Cập nhật trọng số bằng thuật toán AdamW và suy giảm Cosine.
    \end{itemize}
    \item Dừng sớm sau 20 epoch nếu Validation Loss không giảm. Lưu mô hình tối ưu $\theta^*$.
\end{enumerate}
\textbf{Pha 2: Sinh dữ liệu chuỗi thời gian và tiền xử lý:}
\begin{enumerate}
    \item Sử dụng $f_{\theta^*}$ suy luận liên tục trên 608 camera trong 86 ngày để tạo chuỗi 1 phút.
    \item Áp dụng bộ lọc trượt kép nhân quả và lấy mẫu đều đặn $\Delta t = 5$ phút để tạo tensor $\mathbf{Y}_{\text{total}} \in \mathbb{R}^{24768 \times 608 \times 2}$.
    \item Phân chia chuỗi thời gian: $80\%$ Train, $10\%$ Validation, $10\%$ Test.
    \item Chuẩn hóa z-score theo từng nút và đóng băng thông số $\mu_i, \sigma_i$ từ tập Train.
\end{enumerate}
\textbf{Pha 3: Huấn luyện mô hình đồ thị TA-STGCN (Giai đoạn 2):}
\begin{enumerate}
    \item Tính toán ma trận Laplacian chuẩn hóa $\tilde{\mathbf{L}}$ từ ma trận khoảng cách RBF.
    \item Khởi tạo mô hình TA-STGCN với 2 khối ST-Conv và Temporal Multi-Head Attention ($h=4$).
    \item \textbf{Lặp} qua các epoch từ 1 đến 120:
    \begin{itemize}
        \item Lấy batch chuỗi thời gian $B=64$, cửa sổ $T_{\text{in}}=24$, mục tiêu $H=6$.
        \item Lan truyền xuôi qua ST-Conv 1 $\to$ Temporal Attention $\to$ ST-Conv 2 $\to$ Conv1D Output.
        \item Tính hàm mất mát Huber Loss $\mathcal{L}_{\text{Stage 2}}$ và cập nhật trọng số bằng Adam.
        \item Điều chỉnh tốc độ học theo cơ chế ReduceLROnPlateau.
    \end{itemize}
    \item Dừng sớm sau 20 epoch nếu Validation MAE không cải thiện. Lưu trọng số tối ưu $\Phi^*$.
\end{enumerate}
}
\label{algo:training_pipeline}
\end{algorithm_box}

\subsection{Quy trình vận hành suy luận thời gian thực}
\label{sec:quy_trinh_suy_luan}

Để ứng dụng hệ thống vào thực tế giám sát và điều hành giao thông đô thị, quy trình suy luận được tổ chức thành 5 bước liên hoàn:
\begin{enumerate}
    \item \textbf{Bước 1 -- Thu nhận và xử lý ảnh tại camera tuyến đường:} Cứ sau mỗi chu kỳ $\Delta t = 5$ phút, camera trên 608 tuyến đường chụp ảnh hiện trường. Mạng EfficientNet-B5 lượng tử hóa INT8 thực thi suy luận trong khoảng $35$ ms, trích xuất vector số lượng xe tức thời $[\hat{c}_{\text{car}}, \hat{c}_{\text{bike}}]^T$.
    \item \textbf{Bước 2 -- Cập nhật bộ đệm chuỗi thời gian:} Máy chủ trung tâm tiếp nhận 608 vector số đếm (kích thước dữ liệu rất nhỏ gọn), đẩy vào hàng đệm trượt FIFO có độ dài cố định $T_{\text{in}} = 24$ bước (120 phút gần nhất). Điểm dữ liệu cũ nhất tại bước $t-24$ được giải phóng.
    \item \textbf{Bước 3 -- Chuẩn hóa z-score cục bộ:} Dữ liệu trong hàng đệm được chuẩn hóa bằng tham số $\mu_i, \sigma_i$ cố định tương ứng của từng nút: $x_{i, t} = (y_{i, t} - \mu_i) / \sigma_i$.
    \item \textbf{Bước 4 -- Dự báo toàn mạng qua TA-STGCN:} Tensor dữ liệu được đưa qua mô hình TA-STGCN trên GPU máy chủ trung tâm. Thời gian thực thi suy luận toàn mạng chỉ mất đúng \textbf{$4.76$ ms}.
    \item \textbf{Bước 5 -- Giải chuẩn hóa và phát tín hiệu điều phối:} Đầu ra được giải chuẩn hóa về số lượng xe thực tế: $\hat{y}_{i, t+h} = \hat{z}_{i, t+h} \cdot \sigma_i + \mu_i$. Các vector dự báo 6 bước tương lai ($t+5$ đến $t+30$ phút) được cung cấp qua API phục vụ công tác điều phối đèn tín hiệu giao thông và tích hợp lên hệ thống thông tin giám sát đô thị.
\end{enumerate}
